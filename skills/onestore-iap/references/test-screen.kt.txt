/**
 * 원스토어 인앱결제 **동작 확인용 테스트 화면**입니다.
 *
 * [OneStorePurchaseManager] 가 실제로 도는지 눈으로 확인하는 것이 목적입니다 — 로그인 상태,
 * 상품 목록 조회, 결제, 그리고 소비·승인까지 직접 눌러 봅니다.
 *
 * ## 운영 화면이 아닙니다 — 릴리즈 전에 지우거나 `src/debug/` 로 옮기세요
 *
 * 진입점에 연결하지 않았어도 **코드는 배포본에 남습니다.** 확인이 끝났으면 삭제하는 것이 가장
 * 확실하고, 계속 쓰려면 `src/debug/` 소스셋으로 옮겨 릴리즈 빌드에서 빠지게 하세요.
 *
 * - 매니저를 `autoProcessPurchases = false` 로 만듭니다. 구매 후 소비·승인을 **자동으로 하지 않고**
 *   목록에 남겨, 버튼으로 직접 처리해 결과를 보게 합니다
 * - **운영 코드에서는 자동 처리(기본값)를 쓰세요.** 구매 후 3일 안에 소비 또는 승인하지 않으면
 *   미지급으로 판단해 자동 환불됩니다
 * - **상품을 실제로 지급하지 않습니다.** 지급 로직은 운영 코드에 넣으세요
 *
 * ## 확인 순서
 * 1. **로그인** — 원스토어 로그인이 안 돼 있으면 결제가 진행되지 않습니다
 * 2. **상품 목록** — 비어 있으면 `PRODUCT_IDS` 나 개발자센터 판매 상태를 확인하세요
 * 3. **구매** — 상품의 `구매` 를 누릅니다. 테스트 ID 등록 없이 상용 환경에서 누르면 실제 과금됩니다
 * 4. **후처리** — 구매 내역에 뜬 항목을 `소비` 또는 `승인` 으로 처리하고 목록이 갱신되는지 봅니다
 *
 * 소비성 상품(코인 등)은 `소비` 후 재구매가 열리고, 영구성 상품은 `승인` 만 합니다.
 *
 * 두 목록은 남은 높이를 반씩 나눠 씁니다 — 상품이 많아도 「구매 내역」이 화면 밖으로 밀리지
 * 않아야 버튼을 누를 수 있습니다.
 */
@Composable
fun OneStoreTestScreen(activity: Activity, modifier: Modifier = Modifier) {
    var isLoggedIn by remember { mutableStateOf<Boolean?>(null) }
    var manager by remember { mutableStateOf<OneStorePurchaseManager?>(null) }
    var refreshKey by remember { mutableStateOf(0) }

    // 마지막 결과 문구. 리스너 파라미터와 이름이 겹치지 않게 status 로 둡니다.
    var status by remember { mutableStateOf("") }

    val productList = remember { mutableStateListOf<ProductDetail>() }
    val purchaseList = remember { mutableStateListOf<PurchaseData>() }

    // 리스너는 화면 상태만 갱신합니다. 목록 재조회는 refreshKey 가 담당합니다.
    val listener = remember {
        object : OneStorePurchaseManager.Listener {
            override fun onProductsLoaded(products: List<ProductDetail>) {
                productList.clear()
                productList.addAll(products)
                status = "상품 ${products.size}건 조회"
            }

            override fun onPurchaseCompleted(purchase: PurchaseData) {
                status = "후처리 완료: ${purchase.productId}"
                refreshKey++
            }

            override fun onPurchaseCanceled() {
                status = "사용자가 결제를 취소했습니다"
            }

            override fun onError(responseCode: Int, message: String?) {
                status = "오류 $responseCode: ${message ?: ""}"
            }

            override fun onLoginRequired() {
                isLoggedIn = false
                status = "로그인이 필요합니다"
            }

            override fun onUpdateRequired() {
                status = "원스토어 앱 설치·업데이트가 필요합니다"
            }

            override fun onPurchasesNeedProcessing(purchases: List<PurchaseData>) {
                status = "구매 완료 — 소비 또는 승인이 필요합니다 (${purchases.size}건)"
                refreshKey++
            }
        }
    }

    DisposableEffect(Unit) {
        val created = OneStorePurchaseManager(activity, listener, autoProcessPurchases = false)
        manager = created
        created.start()
        created.checkLogin(activity) { isLoggedIn = it }
        refreshKey++

        onDispose {
            created.release()
            manager = null
        }
    }

    // 구매 내역 조회. ProductType.ALL 은 상품 정보 조회 전용이라 타입별로 나눠 부릅니다.
    // AUTO(월정액)는 신규 생성이 중단됐지만 이전에 만든 상품이 남아 있을 수 있어 함께 조회합니다.
    LaunchedEffect(refreshKey) {
        val current = manager ?: return@LaunchedEffect
        purchaseList.clear()
        current.queryProductDetails()
        listOf(ProductType.INAPP, ProductType.SUBS, ProductType.AUTO).forEach { productType ->
            current.queryPurchases(productType) { purchaseList.addAll(it) }
        }
    }

    // 두 목록에 weight 를 주려면 높이가 정해져야 하므로 fillMaxSize 로 받습니다.
    Column(modifier = modifier.fillMaxSize().padding(16.dp)) {
        Text("원스토어 결제 테스트", style = MaterialTheme.typography.titleLarge)

        Row(
            modifier = Modifier.fillMaxWidth().padding(vertical = 8.dp),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            Text(
                text = when (isLoggedIn) {
                    true -> "로그인 됨"
                    false -> "로그인 안 됨"
                    null -> "로그인 확인 중"
                },
                modifier = Modifier.padding(end = 8.dp),
            )
            if (isLoggedIn != true) {
                Button(onClick = { manager?.launchLoginFlow(activity) }) { Text("로그인") }
            }
            OutlinedButton(
                onClick = {
                    manager?.checkLogin(activity) { isLoggedIn = it }
                    refreshKey++
                },
                modifier = Modifier.padding(start = 8.dp),
            ) { Text("새로고침") }
        }

        if (status.isNotEmpty()) {
            Text(status, style = MaterialTheme.typography.bodySmall)
        }

        Text(
            "상품 (${productList.size})",
            style = MaterialTheme.typography.titleMedium,
            modifier = Modifier.padding(top = 16.dp),
        )

        if (productList.isEmpty()) {
            Text(
                "상품이 없습니다 — PRODUCT_IDS 와 개발자센터의 판매 상태를 확인하세요",
                style = MaterialTheme.typography.bodySmall,
            )
        }

        // weight 를 주지 않으면 상품이 많을 때 아래 「구매 내역」이 화면 밖으로 밀려
        // 소비·승인 버튼을 누를 수 없습니다. 두 목록이 남은 높이를 반씩 나눠 씁니다.
        LazyColumn(
            modifier = Modifier.weight(1f),
            verticalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            items(productList) { product ->
                Card(modifier = Modifier.fillMaxWidth()) {
                    Row(
                        modifier = Modifier.fillMaxWidth().padding(12.dp),
                        verticalAlignment = Alignment.CenterVertically,
                    ) {
                        Column(modifier = Modifier.padding(end = 8.dp)) {
                            Text(product.title ?: product.productId)
                            Text(
                                "${product.productId} · ${product.type} · ${product.price}",
                                style = MaterialTheme.typography.bodySmall,
                            )
                        }
                        Button(
                            onClick = {
                                manager?.launchPurchase(activity, product.productId, product.type)
                            },
                        ) { Text("구매") }
                    }
                }
            }
        }

        Text(
            "구매 내역 (${purchaseList.size})",
            style = MaterialTheme.typography.titleMedium,
            modifier = Modifier.padding(top = 16.dp),
        )
        Text(
            "소비는 관리형(INAPP) 상품에만 씁니다. 구독·영구성 상품은 승인만 하세요.",
            style = MaterialTheme.typography.bodySmall,
        )

        LazyColumn(
            modifier = Modifier.weight(1f),
            verticalArrangement = Arrangement.spacedBy(8.dp),
        ) {
            items(purchaseList) { purchase ->
                Card(modifier = Modifier.fillMaxWidth()) {
                    Column(modifier = Modifier.padding(12.dp)) {
                        Text(purchase.productId)
                        Text(
                            "승인됨: ${purchase.isAcknowledged} · 수량 ${purchase.quantity}",
                            style = MaterialTheme.typography.bodySmall,
                        )
                        Row(
                            modifier = Modifier.padding(top = 8.dp),
                            horizontalArrangement = Arrangement.spacedBy(8.dp),
                        ) {
                            Button(onClick = { manager?.consume(purchase) }) { Text("소비") }
                            OutlinedButton(onClick = { manager?.acknowledge(purchase) }) { Text("승인") }
                        }
                    }
                }
            }
        }
    }
}
